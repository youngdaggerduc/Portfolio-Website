function TweaksPanel({tweaks, setTweaks, visible}) {
  if (!visible) return null;
  return (
    <div style={{
      position:'fixed', bottom:24, right:24, zIndex:5000,
      background:'#0d0d2b', border:'3px solid #000',
      boxShadow:'6px 6px 0 #e8192c',
      width:240, overflow:'hidden'
    }}>
      <div style={{background:'var(--red)', padding:'8px 14px', fontFamily:'Bangers', fontSize:'18px', letterSpacing:'3px', borderBottom:'3px solid #000', color:'#fff'}}>TWEAKS</div>
      <div style={{padding:'16px', display:'flex', flexDirection:'column', gap:'14px'}}>
        <div>
          <div style={{fontSize:'10px', letterSpacing:'2px', color:'var(--cyan)', marginBottom:'6px', textTransform:'uppercase'}}>Glitch Effect</div>
          <div style={{display:'flex', gap:'6px'}}>
            {['full','medium','off'].map(v => (
              <button key={v} onClick={() => setTweaks({...tweaks, glitchIntensity:v})}
                style={{
                  flex:1, fontFamily:'Bangers', fontSize:'13px', letterSpacing:'1px',
                  padding:'5px', cursor:'pointer', border:'2px solid',
                  borderColor: tweaks.glitchIntensity===v ? 'var(--red)' : 'rgba(255,255,255,0.2)',
                  background: tweaks.glitchIntensity===v ? 'rgba(232,25,44,0.2)' : 'transparent',
                  color:'var(--white)'
                }}>{v}</button>
            ))}
          </div>
        </div>
        <div>
          <div style={{fontSize:'10px', letterSpacing:'2px', color:'var(--cyan)', marginBottom:'6px', textTransform:'uppercase'}}>Halftone</div>
          <button onClick={() => setTweaks({...tweaks, showHalftone:!tweaks.showHalftone})}
            style={{
              width:'100%', fontFamily:'Bangers', fontSize:'14px', letterSpacing:'2px',
              padding:'6px', cursor:'pointer', border:'2px solid',
              borderColor: tweaks.showHalftone ? 'var(--cyan)' : 'rgba(255,255,255,0.2)',
              background: tweaks.showHalftone ? 'rgba(0,212,255,0.1)' : 'transparent',
              color:'var(--white)'
            }}>{tweaks.showHalftone ? '● ON' : '○ OFF'}</button>
        </div>
      </div>
    </div>
  );
}

function App() {
  const [tweaks, setTweaks] = useState(TWEAK_DEFAULTS);
  const [tweaksVisible, setTweaksVisible] = useState(false);

  useEffect(() => {
    // Apply tweaks
    const root = document.documentElement;
    if (tweaks.glitchIntensity === 'off') {
      document.querySelectorAll('.glitch-layer-1, .glitch-layer-2').forEach(el => el.style.display = 'none');
    } else {
      document.querySelectorAll('.glitch-layer-1, .glitch-layer-2').forEach(el => el.style.display = '');
    }
    if (!tweaks.showHalftone) {
      document.querySelectorAll('.halftone, .halftone-red, .halftone-mag').forEach(el => el.style.backgroundImage = 'none');
    } else {
      document.querySelectorAll('.halftone').forEach(el => el.style.backgroundImage = '');
      document.querySelectorAll('.halftone-red').forEach(el => el.style.backgroundImage = '');
      document.querySelectorAll('.halftone-mag').forEach(el => el.style.backgroundImage = '');
    }
  }, [tweaks]);

  useEffect(() => {
    window.addEventListener('message', e => {
      if (e.data?.type === '__activate_edit_mode') setTweaksVisible(true);
      if (e.data?.type === '__deactivate_edit_mode') setTweaksVisible(false);
    });
    window.parent.postMessage({type:'__edit_mode_available'}, '*');
  }, []);

  const handleTweaks = (t) => {
    setTweaks(t);
    window.parent.postMessage({type:'__edit_mode_set_keys', edits: t}, '*');
  };

  return (
    <>
      <Nav/>
      <HeroSection/>
      <div className="comic-divider"/>
      <AboutSection/>
      <div className="comic-divider" style={{background:'repeating-linear-gradient(90deg, var(--cyan) 0px, var(--cyan) 20px, #000 20px, #000 22px)'}}/>
      <ProjectsSection/>
      <div className="comic-divider" style={{background:'repeating-linear-gradient(90deg, var(--magenta) 0px, var(--magenta) 20px, #000 20px, #000 22px)'}}/>
      <SkillsSection/>
      <div className="comic-divider"/>
      <ContactSection/>
      <footer>
        <div className="footer-logo">PIERCE DOMAN</div>
        <div className="footer-copy">© 2026 — BUILT WITH FULL STACK ENERGY & AI VIBES</div>
      </footer>
      <TweaksPanel tweaks={tweaks} setTweaks={handleTweaks} visible={tweaksVisible}/>
    </>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
